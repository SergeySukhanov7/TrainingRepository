class TestExpResourceTutorial:

    @pytest.mark.usefixtures("tutorial_prepare")
    @pytest.mark.setup_game_settings(standard_exp_settings)
    def test_exp_resource_tutorial_launch(self, launch_control: LaunchControlNew):
        APIExpedition(launch_control.api_client).skip_current_tutorial()

        tutorial_steps = ExpTutorialSteps(launch_control)
        tutorial_steps.prepare_to_check_run_resource_quest_tutorials()
        tutorial_steps.run_resource_tutorial_1()

    @qase.id(17795)
    @pytest.mark.usefixtures("tutorial_prepare")
    @pytest.mark.setup_game_settings(standard_exp_settings)
    def test_exp_resource_tutorial_after_background(self, launch_control: LaunchControlNew):
        APIExpedition(launch_control.api_client).skip_current_tutorial()

        tutorial_steps = ExpTutorialSteps(launch_control)
        tutorial_steps.prepare_to_check_run_resource_quest_tutorials()

        launch_control.driver.background_app(seconds=2)
        time.sleep(1)
        tutorial_steps.run_resource_tutorial_1(need_appear=True)

        launch_control.driver.background_app(seconds=2)
        time.sleep(1)
        tutorial_steps.run_resource_tutorial_2(need_appear=True)

        launch_control.driver.background_app(seconds=2)
        time.sleep(1)
        tutorial_steps.run_resource_tutorial_3(need_appear=True)

        tutorial_steps.run_resource_tutorial_2(need_appear=True)

        tutorial_steps.run_resource_tutorial_3(need_appear=True)
